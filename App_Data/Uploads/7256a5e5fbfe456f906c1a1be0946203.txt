        public void GenerarPDF()
        {
            try
            {
                // Conexión SQL
                using (miConexion)
                {
                    miConexion.Open();

                    // Consulta SQL para obtener datos de la tabla REGISTRO
                    string query = "SELECT * FROM LISTA";
                    string Fecha = DateTime.Now.ToString("d");
                    using (SqlCommand cmd = new SqlCommand(query, miConexion))
                    {
                        using (SqlDataAdapter adapter = new SqlDataAdapter(cmd))
                        {
                            DataTable dataTable = new DataTable();
                            adapter.Fill(dataTable);

                            // Crear documento PDF
                            Document doc = new Document();
                            PdfWriter writer = PdfWriter.GetInstance(doc, new FileStream("LISTA.pdf", FileMode.Create));

                            doc.Open();

                            // Crear tabla en el documento PDF
                            PdfPTable table = new PdfPTable(dataTable.Columns.Count);
                            for (int i = 0; i < dataTable.Columns.Count; i++)
                            {
                                table.AddCell(new Phrase(dataTable.Columns[i].ColumnName));
                            }

                            for (int i = 0; i < dataTable.Rows.Count; i++)
                            {
                                for (int j = 0; j < dataTable.Columns.Count; j++)
                                {
                                    table.AddCell(new Phrase(dataTable.Rows[i][j].ToString()));
                                }
                            }

                            doc.Add(table);

                            doc.Close();
                            writer.Close();
                        }
                    }
                    miConexion.Close();
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show(ex.Message);
            }
        }
